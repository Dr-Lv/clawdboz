# Clawdboz 系统指南

> 本文档是 Clawdboz (嗑唠的宝子) 的全局系统指南，所有 Bot 都会加载此文档作为基础系统提示词。
> 本指南包含了系统架构、API 使用、开发规范等完整信息，是每个 Bot 必须掌握的基础知识。

## 1. 会话目录结构详解

Clawdboz 使用多级 workspace 架构来组织数据，理解这个结构对开发至关重要。

### 1.1 整体架构图

```
项目根目录/                         # 用户项目目录（运行 clawdboz init 的地方）
├── config.json                     # 项目主配置文件
├── .bot.md                         # Bot 专属配置（如果没有就是用默认的）
├── bot0.py                         # Bot 启动脚本
├── bot_manager.sh                  # Bot 管理脚本
├── WORKPLACE/                      # 基础工作目录（所有数据的根）
│   ├── .system.md                  # 本系统指南文件（全局，所有 Bot 共享）
│   ├── workplace_{bot_id}/         # Bot 级 workspace（每个 Bot 一个）
│   │   ├── .bot.md                 # Bot 专属系统提示词
│   │   ├── .agents/                  # Kimi CLI 配置
│   │   │   └── skills/             # Bot 级 skills（所有会话共享）
│   │   │       ├── local-memory/   # 记忆管理 skill
│   │   │       ├── scheduler/      # 定时任务 skill
│   │   │       ├── feishu-api-sender/   # 飞书发送 skill
│   │   │       └── webchat-sender/      # Web Chat 发送 skill
│   │   ├── logs/                   # Bot 级日志
│   │   ├── user_files/             # 用户上传的文件
│   │   ├── user_images/            # 用户上传的图片
│   │   ├── w_{session_id}/         # 单聊会话目录
│   │   │   ├── session.json        # 会话数据（历史记录 + 元数据）
│   │   │   ├── scheduler_tasks.json # 定时任务数据（如适用）
│   │   │   └── logs/               # 会话级日志
│   │   └── g_{session_id}/         # 群聊会话目录（Bot 视角）
│   │       ├── session.json        # 群聊数据（软链接到 groupspace）
│   │       └── scheduler_tasks.json # 群聊定时任务
│   └── groupspace/                 # 共享群聊空间（所有 Bot 共享）
│       └── g_{chat_id}/            # 特定群聊的共享目录
│           ├── session.json        # 群聊数据（所有成员共享同一文件）
│           ├── .members.md         # 群成员信息（自动生成）
│           ├── user_files/         # 群聊文件
│           └── user_images/        # 群聊图片
└── .agents/                          # 项目级 Kimi 配置
    └── skills/                     # 项目级 skills
```

### 1.2 关键目录详解

| 目录/文件 | 用途 | 访问方式 | 重要程度 |
|-----------|------|----------|----------|
| `workplace_{bot_id}/` | Bot 级配置和技能 | 从当前会话目录向上查找 | ⭐⭐⭐ |
| `w_{session_id}/` | 单聊会话数据 | 直接访问当前目录 | ⭐⭐⭐ |
| `g_{session_id}/` | 群聊会话数据（Bot 私有视图） | 直接访问当前目录 | ⭐⭐⭐ |
| `groupspace/g_{chat_id}/` | 共享群聊空间（所有 Bot 共享） | 群聊时的会话目录 | ⭐⭐⭐⭐ |
| `session.json` | 会话数据（历史记录 + 元数据） | 直接读取 JSON | ⭐⭐⭐⭐ |
| `.members.md` | 群成员信息 | 群聊时自动加载到 system prompt | ⭐⭐⭐ |

### 1.4 文件读取方法（重要：只能通过文件读取）

**Bot 只能通过读取文件获取会话信息，不能使用环境变量。**

```python
import json
import os
from typing import Dict, List, Optional

def read_session(session_dir: str = None) -> Dict:
    """
    读取会话数据（历史记录 + 元数据）
    
    Args:
        session_dir: 会话目录路径，默认为当前工作目录
    
    Returns:
        session.json 的完整内容
    """
    if session_dir is None:
        session_dir = os.getcwd()
    
    session_path = os.path.join(session_dir, 'session.json')
    
    if not os.path.exists(session_path):
        return {
            "meta": {"id": "", "chat_type": "single"},
            "messages": [],
            "stats": {"message_count": 0}
        }
    
    with open(session_path, 'r', encoding='utf-8') as f:
        return json.load(f)

def write_session(data: Dict, session_dir: str = None):
    """
    写入会话数据
    
    Args:
        data: 要写入的会话数据
        session_dir: 会话目录路径，默认为当前工作目录
    """
    if session_dir is None:
        session_dir = os.getcwd()
    
    session_path = os.path.join(session_dir, 'session.json')
    os.makedirs(os.path.dirname(session_path), exist_ok=True)
    
    with open(session_path, 'w', encoding='utf-8') as f:
        json.dump(data, f, ensure_ascii=False, indent=2)

def get_messages(session_dir: str = None) -> List[Dict]:
    """获取会话历史消息"""
    session = read_session(session_dir)
    return session.get('messages', [])

def get_meta(session_dir: str = None) -> Dict:
    """获取会话元数据"""
    session = read_session(session_dir)
    return session.get('meta', {})

def get_user_info(session_dir: str = None) -> Dict:
    """获取当前用户信息（单聊时）"""
    meta = get_meta(session_dir)
    return meta.get('user', {})

def get_chat_type(session_dir: str = None) -> str:
    """获取会话类型（single 或 group）"""
    meta = get_meta(session_dir)
    return meta.get('chat_type', 'single')

def is_group_chat(session_dir: str = None) -> bool:
    """判断是否为群聊"""
    return get_chat_type(session_dir) == 'group'

def get_bot_workspace_dir(session_dir: str = None) -> str:
    """
    获取 Bot 级 workspace 目录
    
    从当前会话目录向上查找 workplace_{bot_id}/
    """
    if session_dir is None:
        session_dir = os.getcwd()
    
    current = os.path.abspath(session_dir)
    while current != os.path.dirname(current):  # 防止无限循环
        parent = os.path.dirname(current)
        if os.path.basename(parent).startswith('workplace_'):
            return parent
        current = parent
    
    return None

# 使用示例
if __name__ == "__main__":
    # 读取当前会话数据
    session = read_session()
    
    # 获取消息数量
    messages = session.get('messages', [])
    print(f"当前会话共有 {len(messages)} 条消息")
    
    # 获取用户信息
    user = session.get('meta', {}).get('user', {})
    print(f"用户: {user.get('name', '未知')}")
    
    # 判断会话类型
    if is_group_chat():
        print("当前是群聊")
        members = session.get('meta', {}).get('members', [])
        print(f"成员数: {len(members)}")
    else:
        print("当前是单聊")
```

## 2. 调取记忆 (local-memory) 详解

`local-memory` skill 提供持久化记忆存储，是 Bot 记住用户信息、对话上下文的关键工具。

### 2.1 核心概念

- **记忆**: 一段带有元数据的信息（内容、类别、标签、重要程度等）
- **分类 (category)**: 用于组织记忆的大类别，如 `user_info`, `preference`, `task`, `general`
- **标签 (tags)**: 更细粒度的关键词标记
- **重要程度 (importance)**: 1-5 的评分，用于优先级排序

### 2.2 完整 API 参考

```python
from local_memory import MemoryManager

# 初始化（使用默认路径 - Bot 级 workspace）
memory = MemoryManager()

# ==================== 保存记忆 ====================
memory_id = memory.save(
    content="用户的邮箱是 example@mail.com",
    category="user_info",      # 类别: user_info, preference, task, general
    tags=["email", "contact"], # 标签列表
    importance=5               # 重要程度 1-5
)
# 返回: 唯一记忆 ID（如 "mem_abc123"）

# ==================== 关键词搜索 ====================
results = memory.search("邮箱")
# 返回: [{"id": "...", "content": "...", "category": "...", ...}]

# 指定类别搜索
results = memory.search("邮箱", category="user_info")

# ==================== 语义相似度搜索 ====================
# 基于向量相似度，能找到语义相关但关键词不同的记忆
results = memory.search_similar("联系方式", top_k=5)

# ==================== 获取单个记忆 ====================
mem = memory.get("mem_abc123")

# ==================== 列出所有记忆 ====================
all_memories = memory.list_all()

# 指定类别
user_memories = memory.list_all(category="user_info")

# 限制数量
recent = memory.list_all(limit=20)

# ==================== 更新记忆 ====================
memory.update(
    "mem_abc123",
    content="新的邮箱地址 new@example.com",
    importance=4
)

# ==================== 删除记忆 ====================
memory.delete("mem_abc123")

# ==================== 清理旧记忆 ====================
# 删除 30 天前的记忆
deleted_count = memory.cleanup(days=30)

# ==================== 获取统计信息 ====================
stats = memory.get_stats()
# 返回: {"total": 100, "by_category": {...}, ...}
```

### 2.3 实际使用场景

**场景 1: 记住用户偏好**
```python
from local_memory import MemoryManager

memory = MemoryManager()

# 用户告诉了你他的偏好
memory.save(
    content="用户偏好使用 Python 进行开发，不喜欢 Java",
    category="preference",
    tags=["编程语言", "Python", "Java"],
    importance=4
)

# 稍后用户问技术问题时，先查看他的偏好
results = memory.search("编程语言 preference")
if results:
    print(f"用户偏好: {results[0]['content']}")
    # 在回复时考虑这个偏好
```

**场景 2: 保存任务信息**
```python
# 用户交代了一个任务
memory.save(
    content="用户要求周五前完成报表生成功能的开发",
    category="task",
    tags=["报表", "开发任务", "截止日期"],
    importance=5
)

# 每天检查待办任务
tasks = memory.list_all(category="task")
for task in tasks:
    if "周五" in task['content']:
        print(f"待办任务: {task['content']}")
```

**场景 3: 群聊中的记忆**
```python
# 群聊中记住每个成员的信息
from local_memory import MemoryManager

memory = MemoryManager()

# 保存群成员信息（带群聊标签）
memory.save(
    content=f"用户 @小明 是前端开发工程师，擅长 React",
    category="user_info",
    tags=["群成员", "前端", "React", "小明"],
    importance=4
)

# 搜索群成员信息
members = memory.search("前端工程师", category="user_info")
```

### 2.4 注意事项

1. **存储位置**: 记忆存储在 `workplace_{bot_id}/.agents/skills/local-memory/memory/` 目录下
2. **隔离性**: 每个 Bot 只能访问自己的记忆，无法访问其他 Bot 的记忆
3. **备份**: 重要记忆可以定期导出备份
   ```python
   memory.export("/path/to/backup.json")
   ```
4. **隐私**: 不要保存密码、API Key 等敏感信息到记忆中

## 3. 判断会话类型（群聊 vs 单聊）详解

正确判断会话类型是群聊协作的基础。

### 3.1 判断方法（通过文件读取）

```python
import json
import os

def get_chat_type(session_dir: str = None) -> str:
    """
    判断当前会话类型
    
    通过读取 session.json 的 meta.chat_type 字段
    
    Returns:
        'group' 或 'single'
    """
    if session_dir is None:
        session_dir = os.getcwd()
    
    session_path = os.path.join(session_dir, 'session.json')
    
    if not os.path.exists(session_path):
        # 通过目录特征判断（兼容处理）
        if 'groupspace' in session_dir or '/g_' in session_dir:
            return 'group'
        return 'single'
    
    try:
        with open(session_path, 'r', encoding='utf-8') as f:
            session = json.load(f)
        return session.get('meta', {}).get('chat_type', 'single')
    except Exception:
        return 'single'

def is_group(session_dir: str = None) -> bool:
    """是否为群聊"""
    return get_chat_type(session_dir) == 'group'

def is_single(session_dir: str = None) -> bool:
    """是否为单聊"""
    return get_chat_type(session_dir) == 'single'

# 使用示例
if is_group():
    print("当前是群聊模式")
else:
    print("当前是单聊模式")
```

### 3.2 完整的会话上下文类

```python
import json
import os
from typing import Dict, List, Optional

class SessionContext:
    """会话上下文工具类 - 通过文件读取获取信息"""

    def __init__(self, session_dir: str = None):
        self.session_dir = session_dir or os.getcwd()
        self._session = self._load_session()

    def _load_session(self) -> Dict:
        """加载 session.json"""
        session_path = os.path.join(self.session_dir, 'session.json')
        if os.path.exists(session_path):
            try:
                with open(session_path, 'r', encoding='utf-8') as f:
                    return json.load(f)
            except Exception:
                pass
        return {"meta": {}, "messages": [], "stats": {}}

    @property
    def meta(self) -> Dict:
        """获取元数据"""
        return self._session.get('meta', {})

    @property
    def messages(self) -> List:
        """获取消息列表"""
        return self._session.get('messages', [])

    @property
    def is_group(self) -> bool:
        """是否是群聊"""
        return self.meta.get('chat_type') == 'group'

    @property
    def is_single(self) -> bool:
        """是否是单聊"""
        return not self.is_group

    @property
    def chat_id(self) -> str:
        """获取聊天 ID"""
        return self.meta.get('id', '')

    @property
    def chat_type(self) -> str:
        """获取聊天类型"""
        return 'group' if self.is_group else 'single'

    def get_user_info(self) -> Dict:
        """获取用户信息（单聊时）"""
        return self.meta.get('user', {})

    def get_members(self) -> List:
        """获取成员列表（群聊时）"""
        return self.meta.get('members', [])

    def get_bot_ids(self) -> List[str]:
        """获取参与会话的 Bot ID 列表"""
        return self.meta.get('bot_ids', [])

    def get_name(self) -> str:
        """获取会话名称"""
        return self.meta.get('name', '')

    def get_stats(self) -> Dict:
        """获取会话统计"""
        return self._session.get('stats', {})

# 使用示例
ctx = SessionContext()
print(f"会话类型: {ctx.chat_type}")
print(f"聊天 ID: {ctx.chat_id}")

if ctx.is_group:
    print(f"群成员数: {len(ctx.get_members())}")
    print(f"参与 Bot: {ctx.get_bot_ids()}")
else:
    user = ctx.get_user_info()
    print(f"用户: {user.get('name', '未知')}")
```

## 4. 查看用户介绍详解

用户介绍信息存储在 `session.json` 的 `meta.user` 字段中。

### 4.1 单聊场景完整示例

```python
import json
import os
from typing import Dict, Optional

def get_current_user_info(session_dir: str = None) -> Optional[Dict]:
    """
    获取当前会话的用户信息

    返回:
        {
            "id": "ou_xxx",
            "name": "用户名",
            "intro": "用户自我介绍",
            "avatar": "头像 URL"
        }
    """
    if session_dir is None:
        session_dir = os.getcwd()

    session_path = os.path.join(session_dir, 'session.json')

    if not os.path.exists(session_path):
        return None

    try:
        with open(session_path, 'r', encoding='utf-8') as f:
            session = json.load(f)

        user_info = session.get('meta', {}).get('user', {})

        # 标准化返回
        return {
            'id': user_info.get('id', ''),
            'name': user_info.get('name', '未知用户'),
            'intro': user_info.get('intro', ''),
            'avatar': user_info.get('avatar', ''),
            'chat_type': session.get('meta', {}).get('chat_type', 'single'),
            'chat_id': session.get('meta', {}).get('id', '')
        }

    except Exception as e:
        print(f"读取用户信息失败: {e}")
        return None

# 使用示例
user = get_current_user_info()
if user:
    print(f"正在与 {user['name']} 对话")
    if user['intro']:
        print(f"用户介绍: {user['intro']}")
else:
    print("无法获取用户信息")
```

### 4.2 结合记忆使用

```python
from local_memory import MemoryManager
import json
import os

def get_user_profile(session_dir: str = None):
    """获取完整的用户画像（结合 session.json 和 local-memory）"""
    
    if session_dir is None:
        session_dir = os.getcwd()

    # 1. 从 session.json 获取基础信息
    session_path = os.path.join(session_dir, 'session.json')
    profile = {}

    if os.path.exists(session_path):
        with open(session_path, 'r', encoding='utf-8') as f:
            session = json.load(f)
        user = session.get('meta', {}).get('user', {})
        profile['basic'] = user

    # 2. 从记忆中获取历史信息
    memory = MemoryManager()

    # 搜索所有关于这个用户的记忆
    user_id = profile.get('basic', {}).get('id', '')
    user_name = profile.get('basic', {}).get('name', '')

    # 使用语义搜索查找相关信息
    memories = []
    if user_name:
        memories.extend(memory.search(user_name))

    # 过滤出用户相关的记忆
    user_memories = [
        m for m in memories
        if m.get('category') == 'user_info'
    ]

    profile['memories'] = user_memories

    return profile

# 使用示例
profile = get_user_profile()
print(f"用户: {profile['basic'].get('name')}")
print(f"已记住的信息条数: {len(profile['memories'])}")
for mem in profile['memories']:
    print(f"  - {mem['content']}")
```

### 5.2 读取群成员信息完整代码

```python
import json
import os
from typing import List, Dict

def get_group_members(session_dir: str = None) -> List[Dict]:
    """
    获取群成员列表

    返回:
        [
            {
                "type": "bot",  # 或 "user"
                "id": "bot1",
                "name": "助手一号",
                "intro": "..."
            },
            ...
        ]
    """
    if session_dir is None:
        session_dir = os.getcwd()

    members = []

    # 方法 1: 从 .members.md 读取
    members_md_path = os.path.join(session_dir, '.members.md')
    if os.path.exists(members_md_path):
        with open(members_md_path, 'r', encoding='utf-8') as f:
            content = f.read()
        members = parse_members_md(content)

    # 方法 2: 从 session.json 读取（作为补充）
    session_path = os.path.join(session_dir, 'session.json')
    if os.path.exists(session_path):
        with open(session_path, 'r', encoding='utf-8') as f:
            session = json.load(f)
        meta_members = session.get('meta', {}).get('members', [])
        # 合并（去重）
        existing_ids = {m['id'] for m in members}
        for m in meta_members:
            if m['id'] not in existing_ids:
                members.append(m)

    return members

def parse_members_md(content: str) -> List[Dict]:
    """解析 .members.md 内容"""
    members = []
    current_member = None

    lines = content.split('\n')
    for line in lines:
        line = line.strip()

        # 检测成员标题行: ### @名称 (id)
        if line.startswith('### @'):
            if current_member:
                members.append(current_member)

            # 解析: ### @助手一号 (bot1)
            import re
            match = re.match(r'### @(.+) \((.+)\)', line)
            if match:
                name = match.group(1)
                member_id = match.group(2)
                current_member = {
                    'name': name,
                    'id': member_id,
                    'type': 'unknown',
                    'intro': ''
                }

        # 解析类型
        elif line.startswith('- **类型**:'):
            if current_member:
                current_member['type'] = line.split(':', 1)[1].strip()

        # 解析介绍
        elif line.startswith('- **介绍**:'):
            if current_member:
                current_member['intro'] = line.split(':', 1)[1].strip()

        # 解析技能
        elif line.startswith('- **技能**:'):
            if current_member:
                skills = line.split(':', 1)[1].strip()
                current_member['skills'] = [s.strip() for s in skills.split('、')]

    # 添加最后一个成员
    if current_member:
        members.append(current_member)

    return members

# 使用示例

print(f"群成员总数: {len(members)}")

users = [m for m in members if m['type'] == 'user']
```

## 6. 在群里给其他成员发消息详解

### 6.1 @mention 机制说明

当 Bot 发送的消息中包含 `@bot_name` 时：

```python
import os
import sys
import json
from typing import Dict, Callable

class MessageSender:
    """统一的消息发送器，自动适配场景"""

    def __init__(self):
        self.mode = self._detect_mode()
        self.sender = self._load_sender()

    def _detect_mode(self) -> str:
        """检测当前运行模式"""
        session_dir = os.getcwd()
        
        # 检查目录特征
        if 'groupspace' in session_dir:
            return 'webchat'
        
        # 检查 session.json
        session_path = os.path.join(session_dir, 'session.json')
        if os.path.exists(session_path):
            try:
                with open(session_path, 'r', encoding='utf-8') as f:
                    session = json.load(f)
                bot_ids = session.get('meta', {}).get('bot_ids', [])
                if len(bot_ids) > 1:
                    return 'webchat'
            except:
                pass
        
        # 默认飞书
        return 'feishu'

    def _load_sender(self) -> Dict[str, Callable]:
        """加载对应的发送器"""
        if self.mode == 'webchat':
            return self._load_webchat_sender()
        else:
            return self._load_feishu_sender()

    def _load_webchat_sender(self) -> Dict[str, Callable]:
        """加载 webchat sender"""
        sys.path.insert(0, '.agents/skills/webchat-sender')
        from webchat_sender import send_message, send_file

        return {
            'text': send_message,
            'file': send_file,
            'image': None  # webchat-sender 暂不支持图片
        }

    def _load_feishu_sender(self) -> Dict[str, Callable]:
        """加载 feishu sender"""
        sys.path.insert(0, '.agents/skills/feishu-api-sender')
        from feishu_sender import send_text, send_image, send_file

        return {
            'text': send_text,
            'file': send_file,
            'image': send_image
        }

    def send_text(self, message: str) -> Dict:
        """发送文本消息"""
        return self.sender['text'](message)

    def send_file(self, file_path: str) -> Dict:
        """发送文件"""
        return self.sender['file'](file_path)

    def send_image(self, image_path: str) -> Dict:
        """发送图片（仅飞书支持）"""
        if self.sender['image']:
            return self.sender['image'](image_path)
        else:
            # 不支持图片，转为发送文件
            return self.sender['file'](image_path)
```
